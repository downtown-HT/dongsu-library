2단계: 도서 검색 & 신청서 작성

// 소장도서(구글시트) 검색 기능 async function searchOwnedBooks() { const q = document.getElementById('ownedSearchInput').value.trim(); if (!q) return alert("검색할 책 제목을 입력해주세요."); const resultBox = document.getElementById('ownedResultBox'); const list = document.getElementById('ownedResultList'); resultBox.classList.remove('hidden'); list.innerHTML = `
도서관 서재를 뒤지는 중입니다...
`; try { const res = await fetch(`${GAS_URL}?action=searchOwned&query=${encodeURIComponent(q)}`); const json = await res.json(); if (json.data && json.data.length > 0) { list.innerHTML = json.data.map(b => `
${b.title} (${b.author} | ${b.publisher})
출판년도: ${b.year || '-'} | ISBN: ${b.isbn || '-'}
`).join(''); } else { list.innerHTML = `
🎉 도서관에 없는 책입니다. 아래 2단계에서 신청을 진행해주세요!
`; } } catch (err) { list.innerHTML = `
검색 중 오류가 발생했습니다.
`; } }
